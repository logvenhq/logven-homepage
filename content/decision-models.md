[[extra.sections]]
title = "A practical example"
body = [
  "An operations team could compare alternative maintenance plans against current priorities, available staff and planned downtime. A model may help summarize trade-offs for a planner; the team defines the constraints and makes the decision."
]

[[extra.sections]]
title = "Make the process inspectable"
body = [
  "Sheets can hold the assets, constraints and candidate plans. A worker can apply explicit rules, ask a model to compare options against a defined brief, and record its reasoning and proposed next step for review."
]

[[extra.sections]]
title = "Keep approval with the team"
body = [
  "The workflow can pause for an assigned person to review a recommendation before a consequential change is made. Teams choose which models run, where they run and which steps require approval."
]

[[extra.sections]]
title = "Pilot a decision point"
body = [
  "Choose a recurring decision with known inputs and a clear owner. Use historical or representative cases to see whether the workflow makes assumptions and trade-offs easier to review before connecting it to live work."
]
